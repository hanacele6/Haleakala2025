# 放出から吸着までに複数回バウンドした場合、正味の移動として記録される。
#
# 年ごとのファイル: hop_transport_hist_yrNN.npy (NN は近日点通過ごとの通し番号, 1始まり)

# HOP_TAU_RATIO = 0.0
#   0.0 なら即時マルチバウンドなし = 1ホップの変位
